6: sock_diag(7) Miscellaneous Information Manual sock_diag(7)
7: NAME top
8: sock_diag - obtaining information about sockets
9: SYNOPSIS top
10: #include <sys/socket.h>
11: #include <linux/sock_diag.h>
12: #include <linux/unix_diag.h> /* for UNIX domain sockets */
13: #include <linux/inet_diag.h> /* for IPv4 and IPv6 sockets */
14: diag_socket = socket(AF_NETLINK, socket_type, NETLINK_SOCK_DIAG);
15: DESCRIPTION top
16: The sock_diag netlink subsystem provides a mechanism for obtaining
17: information about sockets of various address families from the
18: kernel. This subsystem can be used to obtain information about
19: individual sockets or request a list of sockets.
20: In the request, the caller can specify additional information it
21: would like to obtain about the socket, for example, memory
22: information or information specific to the address family.
23: When requesting a list of sockets, the caller can specify filters
24: that would be applied by the kernel to select a subset of sockets
25: to report. For now, there is only the ability to filter sockets
26: by state (connected, listening, and so on.)
27: Note that sock_diag reports only those sockets that have a name;
28: that is, either sockets bound explicitly with bind(2) or sockets
29: that were automatically bound to an address (e.g., by connect(2)).
30: This is the same set of sockets that is available via
31: /proc/net/unix, /proc/net/tcp, /proc/net/udp, and so on.
32: Request
33: The request starts with a struct nlmsghdr header described in
34: netlink(7) with nlmsg_type field set to SOCK_DIAG_BY_FAMILY. It
35: is followed by a header specific to the address family that starts
36: with a common part shared by all address families:
37: struct sock_diag_req {
38: __u8 sdiag_family;
39: __u8 sdiag_protocol;
40: };
41: The fields of this structure are as follows:
42: sdiag_family
43: An address family. It should be set to the appropriate
44: AF_* constant.
45: sdiag_protocol
46: Depends on sdiag_family. It should be set to the
47: appropriate IPPROTO_* constant for AF_INET and AF_INET6,
48: and to 0 otherwise.
49: If the nlmsg_flags field of the struct nlmsghdr header has the
50: NLM_F_DUMP flag set, it means that a list of sockets is being
51: requested; otherwise it is a query about an individual socket.
52: Response
53: The response starts with a struct nlmsghdr header and is followed
54: by an array of objects specific to the address family. The array
55: is to be accessed with the standard NLMSG_* macros from the
56: netlink(3) API.
57: Each object is the NLA (netlink attributes) list that is to be
58: accessed with the RTA_* macros from rtnetlink(3) API.
59: UNIX domain sockets
60: For UNIX domain sockets the request is represented in the
61: following structure:
62: struct unix_diag_req {
63: __u8 sdiag_family;
64: __u8 sdiag_protocol;
65: __u16 pad;
66: __u32 udiag_states;
67: __u32 udiag_ino;
68: __u32 udiag_show;
69: __u32 udiag_cookie[2];
70: };
71: The fields of this structure are as follows:
72: sdiag_family
73: The address family; it should be set to AF_UNIX.
74: sdiag_protocol
75: pad These fields should be set to 0.
76: udiag_states
77: This is a bit mask that defines a filter of sockets states.
78: Only those sockets whose states are in this mask will be
79: reported. Ignored when querying for an individual socket.
80: Supported values are:
81: 1 << TCP_ESTABLISHED
82: 1 << TCP_LISTEN
83: udiag_ino
84: This is an inode number when querying for an individual
85: socket. Ignored when querying for a list of sockets.
86: udiag_show
87: This is a set of flags defining what kind of information to
88: report. Each requested kind of information is reported
89: back as a netlink attribute as described below:
90: UDIAG_SHOW_NAME
91: The attribute reported in answer to this request is
92: UNIX_DIAG_NAME. The payload associated with this
93: attribute is the pathname to which the socket was
94: bound (a sequence of bytes up to UNIX_PATH_MAX
95: size).
96: UDIAG_SHOW_VFS
97: The attribute reported in answer to this request is
98: UNIX_DIAG_VFS. The payload associated with this
99: attribute is represented in the following structure:
100: struct unix_diag_vfs {
101: __u32 udiag_vfs_dev;
102: __u32 udiag_vfs_ino;
103: };
104: The fields of this structure are as follows:
105: udiag_vfs_dev
106: The device number of the corresponding on-
107: disk socket inode.
108: udiag_vfs_ino
109: The inode number of the corresponding on-disk
110: socket inode.
111: UDIAG_SHOW_PEER
112: The attribute reported in answer to this request is
113: UNIX_DIAG_PEER. The payload associated with this
114: attribute is a __u32 value which is the peer's inode
115: number. This attribute is reported for connected
116: sockets only.
117: UDIAG_SHOW_ICONS
118: The attribute reported in answer to this request is
119: UNIX_DIAG_ICONS. The payload associated with this
120: attribute is an array of __u32 values which are
121: inode numbers of sockets that has passed the
122: connect(2) call, but hasn't been processed with
123: accept(2) yet. This attribute is reported for
124: listening sockets only.
125: UDIAG_SHOW_RQLEN
126: The attribute reported in answer to this request is
127: UNIX_DIAG_RQLEN. The payload associated with this
128: attribute is represented in the following structure:
129: struct unix_diag_rqlen {
130: __u32 udiag_rqueue;
131: __u32 udiag_wqueue;
132: };
133: The fields of this structure are as follows:
134: udiag_rqueue
135: For listening sockets: the number of pending
136: connections. The size of the array
137: associated with the UNIX_DIAG_ICONS response
138: attribute is equal to this value.
139: For established sockets: the amount of data
140: in incoming queue.
141: udiag_wqueue
142: For listening sockets: the backlog size which
143: equals to the value passed as the second
144: argument to listen(2).
145: For established sockets: the amount of memory
146: available for sending.
147: UDIAG_SHOW_MEMINFO
148: The attribute reported in answer to this request is
149: UNIX_DIAG_MEMINFO. The payload associated with this
150: attribute is an array of __u32 values described
151: below in the subsection "Socket memory information".
152: The following attributes are reported back without any
153: specific request:
154: UNIX_DIAG_SHUTDOWN
155: The payload associated with this attribute is __u8
156: value which represents bits of shutdown(2) state.
157: udiag_cookie
158: This is an array of opaque identifiers that could be used
159: along with udiag_ino to specify an individual socket. It
160: is ignored when querying for a list of sockets, as well as
161: when all its elements are set to -1.
162: The response to a query for UNIX domain sockets is represented as
163: an array of
164: struct unix_diag_msg {
165: __u8 udiag_family;
166: __u8 udiag_type;
167: __u8 udiag_state;
168: __u8 pad;
169: __u32 udiag_ino;
170: __u32 udiag_cookie[2];
171: };
172: followed by netlink attributes.
173: The fields of this structure are as follows:
174: udiag_family
175: This field has the same meaning as in struct unix_diag_req.
176: udiag_type
177: This is set to one of SOCK_PACKET, SOCK_STREAM, or
178: SOCK_SEQPACKET.
179: udiag_state
180: This is set to one of TCP_LISTEN or TCP_ESTABLISHED.
181: pad This field is set to 0.
182: udiag_ino
183: This is the socket inode number.
184: udiag_cookie
185: This is an array of opaque identifiers that could be used
186: in subsequent queries.
187: IPv4 and IPv6 sockets
188: For IPv4 and IPv6 sockets, the request is represented in the
189: following structure:
190: struct inet_diag_req_v2 {
191: __u8 sdiag_family;
192: __u8 sdiag_protocol;
193: __u8 idiag_ext;
194: __u8 pad;
195: __u32 idiag_states;
196: struct inet_diag_sockid id;
197: };
198: where struct inet_diag_sockid is defined as follows:
199: struct inet_diag_sockid {
200: __be16 idiag_sport;
201: __be16 idiag_dport;
202: __be32 idiag_src[4];
203: __be32 idiag_dst[4];
204: __u32 idiag_if;
205: __u32 idiag_cookie[2];
206: };
207: The fields of struct inet_diag_req_v2 are as follows:
208: sdiag_family
209: This should be set to either AF_INET or AF_INET6 for IPv4
210: or IPv6 sockets respectively.
211: sdiag_protocol
212: This should be set to one of IPPROTO_TCP, IPPROTO_UDP, or
213: IPPROTO_UDPLITE.
214: idiag_ext
215: This is a set of flags defining what kind of extended
216: information to report. Each requested kind of information
217: is reported back as a netlink attribute as described below:
218: INET_DIAG_TOS
219: The payload associated with this attribute is a __u8
220: value which is the TOS of the socket.
221: INET_DIAG_TCLASS
222: The payload associated with this attribute is a __u8
223: value which is the TClass of the socket. IPv6
224: sockets only. For LISTEN and CLOSE sockets, this is
225: followed by INET_DIAG_SKV6ONLY attribute with
226: associated __u8 payload value meaning whether the
227: socket is IPv6-only or not.
228: INET_DIAG_MEMINFO
229: The payload associated with this attribute is
230: represented in the following structure:
231: struct inet_diag_meminfo {
232: __u32 idiag_rmem;
233: __u32 idiag_wmem;
234: __u32 idiag_fmem;
235: __u32 idiag_tmem;
236: };
237: The fields of this structure are as follows:
238: idiag_rmem
239: The amount of data in the receive queue.
240: idiag_wmem
241: The amount of data that is queued by TCP but
242: not yet sent.
243: idiag_fmem
244: The amount of memory scheduled for future use
245: (TCP only).
246: idiag_tmem
247: The amount of data in send queue.
248: INET_DIAG_SKMEMINFO
249: The payload associated with this attribute is an
250: array of __u32 values described below in the
251: subsection "Socket memory information".
252: INET_DIAG_INFO
253: The payload associated with this attribute is
254: specific to the address family. For TCP sockets, it
255: is an object of type struct tcp_info.
256: INET_DIAG_CONG
257: The payload associated with this attribute is a
258: string that describes the congestion control
259: algorithm used. For TCP sockets only.
260: pad This should be set to 0.
261: idiag_states
262: This is a bit mask that defines a filter of socket states.
263: Only those sockets whose states are in this mask will be
264: reported. Ignored when querying for an individual socket.
265: id This is a socket ID object that is used in dump requests,
266: in queries about individual sockets, and is reported back
267: in each response. Unlike UNIX domain sockets, IPv4 and
268: IPv6 sockets are identified using addresses and ports. All
269: values are in network byte order.
270: The fields of struct inet_diag_sockid are as follows:
271: idiag_sport
272: The source port.
273: idiag_dport
274: The destination port.
275: idiag_src
276: The source address.
277: idiag_dst
278: The destination address.
279: idiag_if
280: The interface number the socket is bound to.
281: idiag_cookie
282: This is an array of opaque identifiers that could be used
283: along with other fields of this structure to specify an
284: individual socket. It is ignored when querying for a list
285: of sockets, as well as when all its elements are set to -1.
286: The response to a query for IPv4 or IPv6 sockets is represented as
287: an array of
288: struct inet_diag_msg {
289: __u8 idiag_family;
290: __u8 idiag_state;
291: __u8 idiag_timer;
292: __u8 idiag_retrans;
293: struct inet_diag_sockid id;
294: __u32 idiag_expires;
295: __u32 idiag_rqueue;
296: __u32 idiag_wqueue;
297: __u32 idiag_uid;
298: __u32 idiag_inode;
299: };
300: followed by netlink attributes.
301: The fields of this structure are as follows:
302: idiag_family
303: This is the same field as in struct inet_diag_req_v2.
304: idiag_state
305: This denotes socket state as in struct inet_diag_req_v2.
306: idiag_timer
307: For TCP sockets, this field describes the type of timer
308: that is currently active for the socket. It is set to one
309: of the following constants:
310: 0 no timer is active
311: 1 a retransmit timer
312: 2 a keep-alive timer
313: 3 a TIME_WAIT timer
314: 4 a zero window probe timer
315: For non-TCP sockets, this field is set to 0.
316: idiag_retrans
317: For idiag_timer values 1, 2, and 4, this field contains the
318: number of retransmits. For other idiag_timer values, this
319: field is set to 0.
320: idiag_expires
321: For TCP sockets that have an active timer, this field
322: describes its expiration time in milliseconds. For other
323: sockets, this field is set to 0.
324: idiag_rqueue
325: For listening sockets: the number of pending connections.
326: For other sockets: the amount of data in the incoming
327: queue.
328: idiag_wqueue
329: For listening sockets: the backlog size.
330: For other sockets: the amount of memory available for
331: sending.
332: idiag_uid
333: This is the socket owner UID.
334: idiag_inode
335: This is the socket inode number.
336: Socket memory information
337: The payload associated with UNIX_DIAG_MEMINFO and
338: INET_DIAG_SKMEMINFO netlink attributes is an array of the
339: following __u32 values:
340: SK_MEMINFO_RMEM_ALLOC
341: The amount of data in receive queue.
342: SK_MEMINFO_RCVBUF
343: The receive socket buffer as set by SO_RCVBUF.
344: SK_MEMINFO_WMEM_ALLOC
345: The amount of data in send queue.
346: SK_MEMINFO_SNDBUF
347: The send socket buffer as set by SO_SNDBUF.
348: SK_MEMINFO_FWD_ALLOC
349: The amount of memory scheduled for future use (TCP only).
350: SK_MEMINFO_WMEM_QUEUED
351: The amount of data queued by TCP, but not yet sent.
352: SK_MEMINFO_OPTMEM
353: The amount of memory allocated for the socket's service
354: needs (e.g., socket filter).
355: SK_MEMINFO_BACKLOG
356: The amount of packets in the backlog (not yet processed).
357: VERSIONS top
358: NETLINK_INET_DIAG was introduced in Linux 2.6.14 and supported
359: AF_INET and AF_INET6 sockets only. In Linux 3.3, it was renamed
360: to NETLINK_SOCK_DIAG and extended to support AF_UNIX sockets.
361: UNIX_DIAG_MEMINFO and INET_DIAG_SKMEMINFO were introduced in Linux
362: 3.6.
363: STANDARDS top
364: Linux.
365: EXAMPLES top
366: The following example program prints inode number, peer's inode
367: number, and name of all UNIX domain sockets in the current
368: namespace.
369: #include <errno.h>
370: #include <stdio.h>
371: #include <string.h>
372: #include <unistd.h>
373: #include <sys/socket.h>
374: #include <sys/un.h>
375: #include <linux/netlink.h>
376: #include <linux/rtnetlink.h>
377: #include <linux/sock_diag.h>
378: #include <linux/unix_diag.h>
379: static int
380: send_query(int fd)
381: {
382: struct sockaddr_nl nladdr = {
383: .nl_family = AF_NETLINK
384: };
385: struct
386: {
387: struct nlmsghdr nlh;
388: struct unix_diag_req udr;
389: } req = {
390: .nlh = {
391: .nlmsg_len = sizeof(req),
392: .nlmsg_type = SOCK_DIAG_BY_FAMILY,
393: .nlmsg_flags = NLM_F_REQUEST | NLM_F_DUMP
394: },
395: .udr = {
396: .sdiag_family = AF_UNIX,
397: .udiag_states = -1,
398: .udiag_show = UDIAG_SHOW_NAME | UDIAG_SHOW_PEER
399: }
400: };
401: struct iovec iov = {
402: .iov_base = &req,
403: .iov_len = sizeof(req)
404: };
405: struct msghdr msg = {
406: .msg_name = &nladdr,
407: .msg_namelen = sizeof(nladdr),
408: .msg_iov = &iov,
409: .msg_iovlen = 1
410: };
411: for (;;) {
412: if (sendmsg(fd, &msg, 0) < 0) {
413: if (errno == EINTR)
414: continue;
415: perror("sendmsg");
416: return -1;
417: }
418: return 0;
419: }
420: }
421: static int
422: print_diag(const struct unix_diag_msg *diag, unsigned int size)
423: {
424: if (size < NLMSG_LENGTH(sizeof(*diag))) {
425: fputs("short response\n", stderr);
426: return -1;
427: }
428: if (diag->udiag_family != AF_UNIX) {
429: fprintf(stderr, "unexpected family %u\n", diag->udiag_family);
430: return -1;
431: }
432: unsigned int rta_len = size - NLMSG_LENGTH(sizeof(*diag));
433: unsigned int peer = 0;
434: size_t path_len = 0;
435: char path[sizeof(((struct sockaddr_un *) 0)->sun_path) + 1];
436: for (struct rtattr *attr = (struct rtattr *) (diag + 1);
437: RTA_OK(attr, rta_len); attr = RTA_NEXT(attr, rta_len)) {
438: switch (attr->rta_type) {
439: case UNIX_DIAG_NAME:
440: if (!path_len) {
441: path_len = RTA_PAYLOAD(attr);
442: if (path_len > sizeof(path) - 1)
443: path_len = sizeof(path) - 1;
444: memcpy(path, RTA_DATA(attr), path_len);
445: path[path_len] = '\0';
446: }
447: break;
448: case UNIX_DIAG_PEER:
449: if (RTA_PAYLOAD(attr) >= sizeof(peer))
450: peer = *(unsigned int *) RTA_DATA(attr);
451: break;
452: }
453: }
454: printf("inode=%u", diag->udiag_ino);
455: if (peer)
456: printf(", peer=%u", peer);
457: if (path_len)
458: printf(", name=%s%s", *path ? "" : "@",
459: *path ? path : path + 1);
460: putchar('\n');
461: return 0;
462: }
463: static int
464: receive_responses(int fd)
465: {
466: long buf[8192 / sizeof(long)];
467: struct sockaddr_nl nladdr;
468: struct iovec iov = {
469: .iov_base = buf,
470: .iov_len = sizeof(buf)
471: };
472: int flags = 0;
473: for (;;) {
474: struct msghdr msg = {
475: .msg_name = &nladdr,
476: .msg_namelen = sizeof(nladdr),
477: .msg_iov = &iov,
478: .msg_iovlen = 1
479: };
480: ssize_t ret = recvmsg(fd, &msg, flags);
481: if (ret < 0) {
482: if (errno == EINTR)
483: continue;
484: perror("recvmsg");
485: return -1;
486: }
487: if (ret == 0)
488: return 0;
489: if (nladdr.nl_family != AF_NETLINK) {
490: fputs("!AF_NETLINK\n", stderr);
491: return -1;
492: }
493: const struct nlmsghdr *h = (struct nlmsghdr *) buf;
494: if (!NLMSG_OK(h, ret)) {
495: fputs("!NLMSG_OK\n", stderr);
496: return -1;
497: }
498: for (; NLMSG_OK(h, ret); h = NLMSG_NEXT(h, ret)) {
499: if (h->nlmsg_type == NLMSG_DONE)
500: return 0;
501: if (h->nlmsg_type == NLMSG_ERROR) {
502: const struct nlmsgerr *err = NLMSG_DATA(h);
503: if (h->nlmsg_len < NLMSG_LENGTH(sizeof(*err))) {
504: fputs("NLMSG_ERROR\n", stderr);
505: } else {
506: errno = -err->error;
507: perror("NLMSG_ERROR");
508: }
509: return -1;
510: }
511: if (h->nlmsg_type != SOCK_DIAG_BY_FAMILY) {
512: fprintf(stderr, "unexpected nlmsg_type %u\n",
513: (unsigned) h->nlmsg_type);
514: return -1;
515: }
516: if (print_diag(NLMSG_DATA(h), h->nlmsg_len))
517: return -1;
518: }
519: }
520: }
521: int
522: main(void)
523: {
524: int fd = socket(AF_NETLINK, SOCK_RAW, NETLINK_SOCK_DIAG);
525: if (fd < 0) {
526: perror("socket");
527: return 1;
528: }
529: int ret = send_query(fd) || receive_responses(fd);
530: close(fd);
531: return ret;
532: }
