## 3. The drafted no-axis-harm check (run by L1 at measured commit M)

Canonical script (bash, read-only on the lane clone):

```bash
R=/fast/scratch/hyaku/kama-r40
B=efe25f962a0b3a2b9b111c03bb5b5386c9548195
M=<measured full commit>   # per claim receipt

# A1 ancestry: M must descend from B
git -C "$R" merge-base --is-ancestor "$B" "$M" || { echo "FAIL A1 ancestry"; exit 1; }

# A2 changed-path classification
git -C "$R" diff --name-status "$B" "$M" > /tmp/r40v04/changed.tsv
#  - reject any D or R anywhere
#  - reject M outside {SCORECARD.md, LOG.md, corpus/**}
#  - reject A outside {corpus/**, .ufo/waves/round-40/**}
awk -F'\t' '$1=="D"||$1=="R"{print "FAIL A2 " $1 " " $2; f=1}
  $1=="M" && $2 !~ /^(SCORECARD\.md|LOG\.md|corpus\/)/{print "FAIL A2 M " $2; f=1}
  $1=="A" && $2 !~ /^(corpus\/|\.ufo\/waves\/round-40\/)/{print "FAIL A2 A " $2; f=1}
  END{exit f}' /tmp/r40v04/changed.tsv || exit 1

# A3 zero deletions inside any file (no edits to prior lines)
test -z "$(git -C "$R" diff --unified=0 "$B" "$M" | grep -E '^-' | grep -v '^---')" \
  || { echo "FAIL A3 deletions present"; exit 1; }

# A4 strict append-only (KQ7): base blob is a byte-prefix of measured blob
while IFS=$'\t' read -r st path; do
  [ "$st" = A ] && continue
  nb=$(git -C "$R" cat-file -s "$B:$path")
  nm=$(git -C "$R" cat-file -s "$M:$path")
  [ "$nm" -ge "$nb" ] || { echo "FAIL A4 shrink $path"; exit 1; }
  cmp <(git -C "$R" cat-file blob "$B:$path") \
      <(git -C "$R" cat-file blob "$M:$path" | head -c "$nb") \
    || { echo "FAIL A4 prefix $path"; exit 1; }
done < /tmp/r40v04/changed.tsv
echo "admission-safety PASS"
```
