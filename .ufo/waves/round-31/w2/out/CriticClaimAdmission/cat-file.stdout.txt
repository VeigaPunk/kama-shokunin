"""gate-registry-check.py -- falsifier for the 'counter has no gate registry row' mistake class.

Every round counter is a lane instrument; an instrument that SKILL.md never
mentions cannot be graded, re-run or retired by a reader. This check lists the
counters the lane tracks and reports the ones its own SKILL.md does not name.

Contract:
    - The counter population is every tracked path matching
      ``.ufo/waves/round-*/count-*.py`` (one directory level below
      ``.ufo/waves/``, extension ``.py``).
    - The registry is SKILL.md at the evaluated revision. A path is
      ``registered`` iff its repo-relative text occurs verbatim in that file.
    - Each missing path prints ``unregistered <path>`` (sorted). The final line
      is always ``registered R unregistered U``.

Exit codes:
    0  U == 0: every tracked counter is named in SKILL.md
    1  U > 0: at least one tracked counter is not named
    2  could not evaluate (not a git repo, bad revision, missing SKILL.md,
       unreadable file, git failure/timeout)

Read-only: the tool issues only ``git rev-parse --show-toplevel``,
``git ls-tree``, ``git ls-files`` and ``git show``; it never writes to the
repository. That rev-parse is the sole non-show/non-ls-tree command and exists
only to resolve ``--repo`` (default ``.``) and to read the working-tree
SKILL.md at the repository root.

Boundary, stated because it bounds the verdict: this is an occurrence check,
not a table-row check. A path named in prose (e.g. a wave-artifact paragraph)
counts exactly like a registry row, and a registry row's columns -- entry
command, verdict map, self-test, pinned values -- are not inspected. The
round-27 defect requires such a row; this tool can only prove the path is
absent or present.

Usage:
    gate-registry-check.py [--repo DIR] [--rev REV]

    --repo DIR   Repository to evaluate (default: current directory).
    --rev REV    Git revision whose tree and SKILL.md are evaluated.

                 SKILL.md from the filesystem).

from __future__ import annotations

import argparse
import re
import subprocess
import sys
from pathlib import Path

COUNTER_RE = re.compile(r"^\.ufo/waves/round-[^/]+/count-[^/]+\.py$")
SKILL_PATH = "SKILL.md"

EXIT_CLEAN = 0
EXIT_UNREGISTERED = 1
EXIT_ERROR = 2

GIT_TIMEOUT = 30


def run_git(root: str, args: list[str]) -> tuple[int, bytes, str]:
    try:
        p = subprocess.run(
            ["git", "-C", root, *args],
            capture_output=True,
            timeout=GIT_TIMEOUT,
        )
    except (OSError, subprocess.TimeoutExpired) as e:
        return 127, b"", str(e)
    return p.returncode, p.stdout, p.stderr.decode("utf-8", "replace").strip()


def git_root(repo: str) -> tuple[str | None, str]:
    rc, out, err = run_git(repo, ["rev-parse", "--show-toplevel"])
    if rc != 0:
        return None, f"not a git working tree at {repo!r}: {err or f'rc={rc}'}"
    root = out.decode("utf-8", "surrogateescape").strip()
    if not root:
        return None, f"empty repository root for {repo!r}"
    return root, ""


def tracked_counters(root: str, rev: str | None) -> tuple[list[str] | None, str]:
    if rev is None:
        rc, out, err = run_git(root, ["ls-files", "-z"])
    else:
        rc, out, err = run_git(root, ["ls-tree", "-r", "-z", "--name-only", "--full-tree", rev])
    if rc != 0:
        what = "git ls-files" if rev is None else f"git ls-tree {rev}"
        return None, f"{what} failed: {err or f'rc={rc}'}"
    paths = [
        p.decode("utf-8", "surrogateescape")
        for p in out.split(b"\0")
        if p
    ]
    return sorted(p for p in paths if COUNTER_RE.match(p)), ""


def skill_text(root: str, rev: str | None) -> tuple[str | None, str]:
    if rev is None:
        path = Path(root) / SKILL_PATH
        try:
            return path.read_text(encoding="utf-8", errors="surrogateescape"), ""
        except OSError as e:
            return None, f"cannot read {path}: {e}"
    rc, out, err = run_git(root, ["show", f"{rev}:{SKILL_PATH}"])
    if rc != 0:
        return None, f"git show {rev}:{SKILL_PATH} failed: {err or f'rc={rc}'}"
    return out.decode("utf-8", "surrogateescape"), ""


def evaluate(repo: str, rev: str | None) -> tuple[int, list[str], str]:
    root, err = git_root(repo)
    if root is None:
        return EXIT_ERROR, [], err
    counters, err = tracked_counters(root, rev)
    if counters is None:
        return EXIT_ERROR, [], err
    text, err = skill_text(root, rev)
    if text is None:
        return EXIT_ERROR, [], err

    unregistered = [p for p in counters if p not in text]
    registered = len(counters) - len(unregistered)
    lines = [f"unregistered {p}" for p in unregistered]
    lines.append(f"registered {registered} unregistered {len(unregistered)}")
    return (EXIT_CLEAN if not unregistered else EXIT_UNREGISTERED), lines, ""


def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(
        description="Report tracked round counters that SKILL.md does not name.",
    )
    ap.add_argument("--repo", default=".", help="repository to evaluate (default: .)")
    ap.add_argument(
        "--rev",
        default=None,
        help="git revision to evaluate (default: the working tree)",
    )
    args = ap.parse_args(argv)

    code, lines, err = evaluate(args.repo, args.rev)
    if code == EXIT_ERROR:
        print(f"ERROR could not evaluate: {err}", file=sys.stderr)
        return EXIT_ERROR
    for line in lines:
        print(line)
    return code


if __name__ == "__main__":
    sys.exit(main())

def f32_eq(a, b) -> bool:
    if isinstance(a, bool) or isinstance(b, bool):
        return a == b
    if isinstance(a, (int, float)) and isinstance(b, (int, float)):
        try:
            return abs(float(a) - float(b)) <= 1e-6 * max(1.0, abs(float(a)), abs(float(b)))
        except OverflowError:
            return False
    return a == b


def normalize_response(body: bytes) -> dict:
    try:
        d = json.loads(body.decode())
    except Exception:
        return {"raw_prefix": body[:40].decode(errors="replace")}
    choices = d.get("choices") or []
    return {
        "top_keys": sorted(d.keys()),
        "n_choices": len(choices),
        "choice_keys": [sorted(c.keys()) for c in choices if isinstance(c, dict)],
        "has_logprobs": bool(choices) and isinstance(choices[0], dict)
                        and choices[0].get("logprobs") is not None,
        "usage_keys": sorted((d.get("usage") or {}).keys()),
        "prompt_tokens": (d.get("usage") or {}).get("prompt_tokens"),
    }


def render(row: dict) -> str:
    o = row["outcome"]

def git(repo, *args, check=True):
    env = {k: v for k, v in os.environ.items()
           if k not in ("GIT_DIR", "GIT_WORK_TREE", "GIT_INDEX_FILE", "GIT_OBJECT_DIRECTORY",
                        "GIT_ALTERNATE_OBJECT_DIRECTORIES", "GIT_REPLACE_REF_BASE")}

class Repo:
    def __init__(self, path):
        self.path = path
        self._blob = {}
        self._anc = {}

    def blob(self, rev, path):
        key = (rev, path)
        if key not in self._blob:
            p = git(self.path, "cat-file", "blob", f"{rev}:{path}", check=False)
            self._blob[key] = p.stdout if p.returncode == 0 else None
        return self._blob[key]

    def commit(self, token):
        p = git(self.path, "rev-parse", "--verify", "--quiet", f"{token}^{{commit}}", check=False)
        return p.stdout.decode().strip() if p.returncode == 0 else None

    def ancestor(self, a, w):
        if a == w:
            return True
        key = (a, w)
        if key not in self._anc:
            self._anc[key] = git(self.path, "merge-base", "--is-ancestor", a, w,
                                 check=False).returncode == 0
        return self._anc[key]

    lines = text.split("\n")
    base_raw = repo.blob(BASE, "SCORECARD.md")
    append_only = base_raw is not None and raw.startswith(base_raw)

    # Split off the appended correction section.
    corr_start = None
    for idx, ln in enumerate(lines):
        if ln.startswith(CORR_HEADING):
            corr_start = idx
            break
    main_lines = lines if corr_start is None else lines[:corr_start]
    corr_lines = [] if corr_start is None else lines[corr_start:]

    problems = []      # list of dicts
    rows_checked = 0
    exempt = []
    for start, heading, hdr, rows in parse_tables(main_lines):
        if not hdr or hdr[0].strip().lower() != "claim":
            continue
        m = RE_SECTION.match(heading)
        rnd = m.group(1) if m else None
        hdr_l = [c.lower() for c in hdr]
        for li, cs in rows:
            if not cs or not cs[0].strip():
                continue
            r = dict(zip(hdr_l, cs))
            rid = cs[0].strip()
            row_text = " | ".join(cs)
            rows_checked += 1
            cmd = r.get("counting command", "") or r.get("counting commands", "")
            if "(exploratory)" in rid.lower() and norm(cmd).lower().startswith("none"):
                exempt.append(rid)
                continue
            rev = None
            pin_text = r.get("measured commit", "") or row_text
            for tok in RE_HEX.findall(pin_text) + RE_HEX.findall(row_text):
                c = repo.commit(tok)
                if c and repo.ancestor(c, tip):
                    rev = c
                    break
            if rev is None:
                problems.append({"row": rid, "line": li + 1, "round": rnd, "class": "p1",
                                 "observed": "p1:no-revision"})
                continue
            def resolve(tok, where):
                cands = [tok]
                if rnd:
                    cands.append(f".ufo/waves/round-{rnd}/{tok}")
                for cand in cands:
                    if repo.blob(where, cand) is not None:
                        return cand
                return None

            paths = [t for t in RE_PATH.findall(row_text)
                     if not (t.startswith("../") or t.startswith("/"))]
            if not any(resolve(t, tip) for t in paths):
                problems.append({"row": rid, "line": li + 1, "round": rnd, "class": "p2",
                                 "observed": "p2:no-evidence-path"})
                continue
            bad = None
            scripts = [t for t in RE_PATH.findall(cmd) if t.endswith(".py")]
            for script in scripts:
                if resolve(script, tip) is None:
                    bad = {"row": rid, "line": li + 1, "round": rnd, "class": "p3",
                           "observed": f"p3:counter-absent:{script}"}
                    break
            if bad is None:
                for pth, algo, dig in RE_DIGEST.findall(row_text):
                    res = resolve(pth, rev)
                    if res is None:
                        continue
                    blob = repo.blob(rev, res)
                    actual = hashlib.sha256(blob).hexdigest() if algo == "sha256" \
                        else hashlib.md5(blob).hexdigest()
                    if not actual.startswith(dig) and not dig.startswith(actual[:len(dig)]):
                        bad = {"row": rid, "line": li + 1, "round": rnd, "class": "p3",
                               "observed": f"p3:digest-mismatch:{res}={actual}"}
                        break
            if bad:
                problems.append(bad)

    # Parse corrections.
    corrections = []
    for start, heading, hdr, rows in parse_tables(corr_lines):
        if not heading.startswith(CORR_HEADING):
            continue
        if [h.strip().lower() for h in hdr] != CORR_HEADER:
            print(f"cannot evaluate: correction table header {hdr!r} != {CORR_HEADER!r}",
                  file=sys.stderr)
            return 2
        for li, cs in rows:
            if len(cs) != 4:
                corrections.append({"id": cs[0] if cs else "", "row": "", "class": "",
                                    "observed": "", "ok": False})
                continue
            corrections.append({"id": cs[0], "row": cs[1], "class": cs[2].strip(),
                                "observed": cs[3].strip(), "ok": None})

    by_key = {}
    for c in corrections:
        key = (c["row"].strip(), c["class"])
        by_key.setdefault(key, []).append(c)
    false_corrections = 0
    valid_map = {}
    problem_by_key = {(p["row"], p["class"]): p for p in problems}
    for c in corrections:
        key = (c["row"].strip(), c["class"])
        p = problem_by_key.get(key)
        ok = (p is not None and len(by_key[key]) == 1
              and c["observed"] == p["observed"] and c["id"].startswith("CR31:"))
        c["ok"] = ok
        if ok:
            valid_map[key] = c
        else:
            false_corrections += 1
    unresolved = [p for p in problems if (p["row"], p["class"]) not in valid_map]
    valid = len(valid_map)
    met = (rows_checked >= 28 and not unresolved and false_corrections == 0 and append_only)

    summary = (f"rows {rows_checked} base_mismatched {len(problems)} exempt {len(exempt)} "
               f"valid_corrections {valid} false_corrections {false_corrections} "
               f"unresolved {len(unresolved)} append_only {append_only} met {met} tip {tip[:12]}")
    print(summary)
    if a.list:
        for p in problems:
            print(f"  {p['row']}: {p['class']} line {p['line']} {p['observed']}")
        for c in corrections:
            print(f"  corr {c['id']} row={c['row']} class={c['class']} ok={c['ok']}")
    if a.out:
        with open(a.out, "w") as fh:
            json.dump({"summary": summary, "tip": tip, "base": BASE,
                       "rows": rows_checked, "exempt": exempt,
                       "problems": problems, "corrections": corrections,
                       "valid_corrections": valid, "false_corrections": false_corrections,
                       "unresolved": [p["row"] for p in unresolved],
                       "append_only": append_only, "met": met, "list": a.list}, fh, indent=1)
            fh.write("\n")
    return 0


if __name__ == "__main__":
    sys.exit(main())
