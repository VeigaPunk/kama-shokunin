def sh(argv, check=True):
    r = subprocess.run(argv, capture_output=True, text=True)
    if check and r.returncode != 0:
        raise RuntimeError("{} -> {}\n{}\n{}".format(" ".join(argv), r.returncode, r.stdout, r.stderr))
    return r


def git(d, *args, check=True):
    return sh(["git", "-C", d, *GITID, *args], check=check)


def rev(d):
    return git(d, "rev-parse", "HEAD").stdout.strip()


def put(d, rel, text):
    p = os.path.join(d, rel)
    os.makedirs(os.path.dirname(p) or d, exist_ok=True)
    with open(p, "w", encoding="utf-8") as fh:
        fh.write(text)


def init(path):
    os.makedirs(path, exist_ok=True)
    git(path, "init", "-q", "-b", "master")
    top = os.path.realpath(git(path, "rev-parse", "--show-toplevel").stdout.strip())
    assert top == os.path.realpath(path), "toplevel {} != fixture {}".format(top, path)
    return path


def commit(d, msg):
    git(d, "add", "-A")
    git(d, "commit", "-q", "-m", msg)
    return rev(d)

# ------------------------------------------------------- reference claim pass


def load_base():
    spec = importlib.util.spec_from_file_location("base_binding", BASE_COPY)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod


def reference_claim(repo, tip, rnd, mod):
    """Frozen C2 claim sentence, implemented in scratch over the base tool's readers."""
    tip_sha = mod._commit_sha(repo, tip)
    sc = mod._blob(repo, tip_sha, "SCORECARD.md")
    lesson_path = "corpus/lessons/round-{}.md".format(rnd)
    lesson = mod._blob(repo, tip_sha, lesson_path)
    if sc is None or lesson is None:
        return 2, ["UNREADABLE self-record"], 0, 0
    sec = mod._round_section(sc.decode("utf-8", "replace"), rnd)
    if sec is None:
        return 2, ["UNREADABLE no '## Round {}' section".format(rnd)], 0, 0
    lines, offset = sec
    sc_lines = sc.decode("utf-8", "replace").splitlines()
    ls_lines = lesson.decode("utf-8", "replace").splitlines()
    cites = list(mod._citations(lines, offset, "SCORECARD.md"))
    cites += list(mod._citations(ls_lines, 0, lesson_path))
    by_line = {}
    for token, filename, lineno in cites:
        try:
            paths = mod._resolve_citation(repo, tip_sha, rnd, token,
                                          "{}:{}".format(filename, lineno)) or []
        except mod.CouldNotEvaluate as exc:
            return 2, ["UNREADABLE {}".format(exc)], 0, 0
        if paths:
            by_line.setdefault((filename, lineno), []).extend(paths)
    claims = 0
    unbound = 0
    findings = []
    for (filename, lineno), paths in sorted(by_line.items()):
        src = sc_lines if filename == "SCORECARD.md" else ls_lines
        line = src[lineno - 1].lower()
        if not any(p in line for p in PHRASES):
            continue
        claims += 1
        bad = []
        for path in sorted(set(paths)):
            blob = mod._blob(repo, tip_sha, path)
            if blob is None:
                bad.append((path, "absent"))
                continue
            hrev, problem = mod._header_rev(blob)
            if hrev is None:
                bad.append((path, problem))
            elif hrev != tip_sha:
                bad.append((path, "header {} != tip {}".format(hrev, tip_sha)))
        if bad:
            unbound += 1
            for path, why in bad:
                findings.append("RECEIPT_CLAIM_FINDING stale-claim {} :: {} :: {}".format(
                    path, why, "{}:{}".format(filename, lineno)))
    return (0 if not findings else 1), findings, claims, unbound


def line(label, rc, out, claim_rc=None, claim_out=None, claims=None, unbound_claims=None):
    print("== {}".format(label))
    print("   base exit={} :: {}".format(rc, out.replace("\n", " | ")))
    if claim_rc is not None:
        print("   claim exit={} :: {}".format(claim_rc, (claim_out or "").replace("\n", " | ")))
    if claims is not None:
        print("   reference claims={} unbound_claims={}".format(claims, unbound_claims))

"""Landing-sequence simulation on a --shared clone of the lane (read-only over the lane).

Rebuilds the planned sequence -- instrument tip T, receipts stamped # rev=T, record commit R,
final binding run at R -- then breaks it in the four ways the plan can still lose:
  A  record cites the pre-freeze control receipt (header 6e961996, delta includes the freeze
     commit that wrote plans/round-73-claims.md)
  B  a seat receipt stamped at the pre-instrument master bc9e3059
  C  an instrument commit lands AFTER T (SKILL.md registry rows) and the run moves to the new tip
  D  T is amended after the receipts name it
  E  the record commit itself carries one instrument file (corpus/QUALITY.md criterion)
"""

def sh(argv, check=True):
    r = subprocess.run(argv, capture_output=True, text=True)
    if check and r.returncode != 0:
        raise RuntimeError("{} -> {}\n{}\n{}".format(" ".join(argv), r.returncode, r.stdout, r.stderr))
    return r


def git(d, *args, check=True):
    return sh(["git", "-C", d, *GITID, *args], check=check)


def rev(d, ref="HEAD"):
    return git(d, "rev-parse", ref).stdout.strip()


def put(d, rel, text):
    p = os.path.join(d, rel)
    os.makedirs(os.path.dirname(p) or d, exist_ok=True)
    with open(p, "w", encoding="utf-8") as fh:
        fh.write(text)


def commit(d, msg):
    git(d, "add", "-A")
    git(d, "commit", "-q", "-m", msg)

def base(repo, tip="master", rnd=73):

def show(label, rc, out):
    print("== {}".format(label))
    for ln in out.splitlines():
        print("   " + ln)
    print("   exit={}".format(rc))


def fresh(name):
    d = os.path.join(WORK, name)
    if os.path.exists(d):
        shutil.rmtree(d)
    sh(["git", "clone", "-q", "--shared", LANE, d])
    top = os.path.realpath(git(d, "rev-parse", "--show-toplevel").stdout.strip())
    assert top == os.path.realpath(d), "{} not a git root".format(d)
    return d

"""Wording classes: which tip-claim spellings the frozen C2 claim pass sees, and which it misses.

def sh(argv, check=True):
    r = subprocess.run(argv, capture_output=True, text=True)
    if check and r.returncode != 0:
        raise RuntimeError("{} -> {}\n{}\n{}".format(" ".join(argv), r.returncode, r.stdout, r.stderr))
    return r


def git(d, *args):

def put(d, rel, text):
    p = os.path.join(d, rel)
    os.makedirs(os.path.dirname(p) or d, exist_ok=True)
    with open(p, "w", encoding="utf-8") as fh:

            ("at tip (bare)",
             "the check at tip: `measured/after/g.out`."),
        ]
        for i, (_label, _line) in enumerate(rows):
            put(d, ".ufo/waves/round-73/measured/after/{}.out".format(chr(97 + i)),
                "# rev={}\nverdict green\n".format(t0))
        body = "# s\n\n## Round 72 — p\n\nold\n\n## Round 73 — current\n\n"

#!/usr/bin/env python3
"""Write the CRT02RecordCritic receipts.

Each receipt's first line is `# rev=<git -C LANE rev-parse master>` read at write time; the body
is the verbatim stdout+stderr of a command this script actually runs.
"""

from __future__ import annotations

import os
import shutil
import subprocess
import sys

S = os.path.dirname(os.path.abspath(__file__))
LANE = "/home/vgpnk/Projects/shokunin/kama-shokunin"
W = os.path.join(LANE, ".ufo/waves/round-73/w1/CRT02RecordCritic")
PY = os.environ.get("PYTHONDONTWRITEBYTECODE", "1")


def sh(argv, cwd=S):
    env = dict(os.environ)
    env["PYTHONDONTWRITEBYTECODE"] = PY
    r = subprocess.run(argv, cwd=cwd, capture_output=True, text=True, env=env)
    return r.returncode, r.stdout + r.stderr


def master():
    return sh(["git", "-C", LANE, "rev-parse", "master"])[1].strip()


def write(name, cmdline, rc, body, extra=""):
    rev = master()
    path = os.path.join(W, name)
    with open(path, "w", encoding="utf-8") as fh:
        fh.write("# rev={}\n".format(rev))
        fh.write("# command: {}\n".format(cmdline))
        if extra:
            fh.write("# note: {}\n".format(extra))
        fh.write(body)
        if not body.endswith("\n"):
            fh.write("\n")
        fh.write("# exit={}\n".format(rc))
    print("wrote {} rev={} exit={}".format(path, rev, rc))


def main():
    os.makedirs(W, exist_ok=True)

    # instrument identity at measurement time
    rc, out = sh(["sha256sum",
                  os.path.join(LANE, "tools/round73-round-scope-cases.py"),
                  os.path.join(LANE, "tools/round73-receipt-claim-binding.py"),
                  os.path.join(LANE, "tools/round73-province-host-drift.py"),
                  os.path.join(LANE, "tools/round72-receipt-tip-binding.py"),
                  os.path.join(LANE, "tools/pre-report-gate.py")])
    print("instrument sha256:\n" + out)

    for name, script in (("fixtures.out", "drive.py"),
                         ("sequence.out", "seq.py"),
                         ("wording.out", "wording.py"),
                         ("realhistory.out", "realhistory.py")):
        rc, body = sh([sys.executable, os.path.join(S, script)])
        write(name, "python3 {} (cwd {})".format(script, S), rc, body,
              extra="instrument sha256:\n" + out.strip())

    # arm 6 exactly as tools/pre-report-gate.py builds it
    arm = [sys.executable, os.path.join(LANE, "tools/round73-round-scope-cases.py"),
           "--repo", os.path.realpath(LANE), "--round", "73", "--tip", "master"]
    rc, body = sh(arm)
    clean_argv = ["git", "-C", LANE, "status", "--porcelain", "-uall", "--", ".ufo/waves/round-73"]
    rcc, clean_out = sh(clean_argv)
    records = [ln for ln in clean_out.splitlines() if ln.strip()]
    classes = {}
    for ln in body.splitlines():
        if ln.startswith("ROUND_SCOPE_V2_FINDING "):
            classes[ln.split()[1]] = classes.get(ln.split()[1], 0) + 1
    summary = ("\n# finding classes: {}\n"
               "# round-clean arm (gate argv verbatim): {} -> files={}\n").format(
        classes, " ".join(clean_argv), len(records))
    write("arm6-live.out",
          " ".join(arm) + " ; and " + " ".join(clean_argv),
          rc, body.rstrip("\n") + summary)

    claim = [sys.executable, os.path.join(LANE, "tools/round73-receipt-claim-binding.py"),
             "--repo", os.path.realpath(LANE), "--round", "73", "--tip", "master"]
    rc, body = sh(claim)
    shutil.rmtree(os.path.join(S, "selftest-c2b"), ignore_errors=True)
    rcs, selftest = sh(claim + ["--selftest", "--work", os.path.join(S, "selftest-c2b")])
    write("claim-live.out",
          " ".join(claim) + " ; then the same argv with --selftest --work <scratch>",
          rc, body.rstrip("\n") + "\n\n== selftest rc={}\n".format(rcs) +
          "\n".join(selftest.strip().splitlines()[-8:]))

    shutil.rmtree(os.path.join(S, "selftest-c1b"), ignore_errors=True)
    c1 = [sys.executable, os.path.join(LANE, "tools/round73-round-scope-cases.py"),
          "--selftest", "--work", os.path.join(S, "selftest-c1b")]
    rc, body = sh(c1, cwd=LANE)
    tail = body.strip().splitlines()
    write("c1-selftest.out", "cd {} && ".format(LANE) + " ".join(c1), rc,
          "\n".join(tail[-30:]) + "\n# full output lines={}\n".format(len(tail)),
          extra="run from the lane cwd: the module's --selftest plan-predicate arm resolves "
                "--repo '.' (its default) and exits 1 outside a git repository")

    return 0


if __name__ == "__main__":
    sys.exit(main())
